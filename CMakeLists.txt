cmake_minimum_required(VERSION 3.22)
project(GodOfWar2Android LANGUAGES C CXX)
if(NOT ANDROID)
    message(FATAL_ERROR "Use the Android NDK toolchain. See tools/android/build.sh")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(PS2X_BUILD_RECOMP OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_ANALYZER OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(PS2X_BUILD_STUDIO OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_DEBUG_UI OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_AGRESSIVE_LOGS OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_IOP_RPC_TRACE OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_RUNNER_UNITY_BUILD OFF CACHE BOOL "" FORCE)
set(PS2X_ENABLE_RUNNER_PCH OFF CACHE BOOL "" FORCE)
set(OPENGL_VERSION "ES 3.0" CACHE STRING "" FORCE)
set(PS2X_DEFAULT_BOOT_ELF "game/SCUS_974.81" CACHE STRING "" FORCE)
if(ANDROID_ABI STREQUAL "x86_64")
    add_compile_options(-msse4.1)
endif()
add_compile_options(-fno-strict-aliasing)
set(GOW_GENERATED_DIR "${CMAKE_CURRENT_SOURCE_DIR}/generated" CACHE PATH "Locally generated PS2 game sources")
if(NOT EXISTS "${GOW_GENERATED_DIR}/output/ps2_recompiled_functions.h")
    message(FATAL_ERROR "Run tools/android/prepare.py with your local USA v1.01 disc before building. No game code is bundled.")
endif()
add_subdirectory(PS2Recomp)
file(GLOB GOW_EE_SOURCES CONFIGURE_DEPENDS "${GOW_GENERATED_DIR}/output/*.cpp")
file(GLOB GOW_NATIVE_SOURCES CONFIGURE_DEPENDS "${GOW_GENERATED_DIR}/native/*.cpp")
target_sources(ps2EntryRunner PRIVATE ${GOW_EE_SOURCES} ${GOW_NATIVE_SOURCES} overrides/gow2_overrides.cpp native/android_bridge.cpp)
target_include_directories(ps2EntryRunner PRIVATE "${GOW_GENERATED_DIR}/output" "${GOW_GENERATED_DIR}/native" "${CMAKE_CURRENT_SOURCE_DIR}/native" "${CMAKE_CURRENT_SOURCE_DIR}/PS2Recomp/ps2xIOP/src")
target_include_directories(ps2_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native")
target_link_libraries(ps2_runtime PUBLIC EGL GLESv3 dl)
# Keep the game-registration constructors in the shared library.
target_link_options(ps2EntryRunner PRIVATE "-Wl,-z,max-page-size=16384")

set_target_properties(ps2EntryRunner PROPERTIES UNITY_BUILD ON UNITY_BUILD_BATCH_SIZE 32)
set_source_files_properties(${GOW_NATIVE_SOURCES} overrides/gow2_overrides.cpp native/android_bridge.cpp
    "${CMAKE_CURRENT_SOURCE_DIR}/PS2Recomp/ps2xRuntime/src/main.cpp" TARGET_DIRECTORY ps2EntryRunner PROPERTIES SKIP_UNITY_BUILD_INCLUSION ON)

# Preserve the VU multiply/add rounding boundaries when translating SSE to NEON.
set_source_files_properties(${GOW_NATIVE_SOURCES} TARGET_DIRECTORY ps2EntryRunner
    PROPERTIES COMPILE_OPTIONS "-ffp-contract=off")

# All game/runtime calls resolve inside this self-contained native library.
target_link_options(ps2EntryRunner PRIVATE "-Wl,-Bsymbolic-functions")
